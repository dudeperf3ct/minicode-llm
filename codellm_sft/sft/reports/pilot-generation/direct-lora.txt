def histogram_equalization(image):
    if not image:
        return []
    if not image[0]:
        return []

    # Flatten the image to a list of pixel values
    pixels = [pixel for row in image for pixel in row]

    # Count the frequency of each pixel value
    freq = {}
    for p in pixels:
        freq[p] = freq.get(p, 0) + 1

    # Calculate the total number of pixels
    total = len(pixels)

    # Calculate the cumulative distribution function (CDF)
    cdf = {}
    sorted_keys = sorted(freq.keys())
    cumulative = 0
    for key in sorted_keys:
        cumulative += freq[key]
        cdf[key] = cumulative

    # Calculate the new pixel values
    new_pixels = []
    for p in pixels:
        if p in cdf:
            new_pixels.append(cdf[p])

    # Reshape the new pixels back into the original image dimensions
    new_image = []
    for i in range(0, len(new_pixels), len(image[0])):
        new_image.append(new_pixels[i:i + len(image[0])])

    return new_image

def evaluateExpression(expression: str) -> int:
    stack = []
    current = 0
    sign = 1
    for char in expression:
        if char.isdigit():
            current = current * 10 + int(char)
        elif char in '+-':
            stack.append(sign * current)
            current = 0
            sign = 1 if char == '+' else -1
        elif char == '(':
            stack.append(current)
            current = 0
        elif char == ')':
            stack.append(current)
            current = 0
    stack.append(sign * current)
    result = 0
    while stack:
        result += stack.pop()
    return result

def trap(height):
    if not height:
        return 0
    n = len(height)
    left_max = [0] * n
    right_max = [0] * n
    left_max[0] = height[0]
    for i in range(1, n):
        left_max[i] = max(left_max[i-1], height[i])
    right_max[-1] = height[-1]
    for i in range(n-2, -1, -1):
        right_max[i] = max(right_max[i+1], height[i])
    total = 0
    for i in range(n):
        total += min(left_max[i], right_max[i]) - height[i]
    return total

def count_distinct_values(test_cases):
    return [len(set(case)) for case in test_cases]

def parse_input(input_string):
    lines = input_string.strip().split('\n')
    n = int(lines[0])
    test_cases = []
    for i in range(1, n + 1):
        case = list(map(int, lines[i].split()))
        test_cases.append(case)
