def histogram_equalization(image):
    if not image:
        return []
    rows = len(image)
    cols = len(image[0])
    flat = [pixel for row in image for pixel in row]
    sorted_flat = sorted(flat)
    unique_values = sorted(set(flat))
    if not unique_values:
        return image
    total = len(flat)
    new_values = []
    for val in unique_values:
        count = flat.count(val)
        new_val = int((count / total) * 255)
        new_values.append(new_val)
    new_image = []
    for row in image:
        new_row = [new_values[flat.index(pixel)] for pixel in row]
        new_image.append(new_row)
    return new_image

def evaluateExpression(expression: str) -> int:
    stack = []
    current = 0
    sign = 1
    for char in expression:
        if char.isdigit():
            current = current * 10 + int(char)
        else:
            if char == '(':
                stack.append((sign, current))
                current = 0
                sign = 1
            elif char == ')':
                stack.append((sign, current))
                current = 0
                sign = 1
            elif char == '+':
                stack.append((sign, current))
                current = 0
                sign = 1
            elif char == '-':
                stack.append((sign, current))
                current = 0
                sign = -1
    stack.append((sign, current))
    result = 0
    for sign, num in stack:
        result += sign * num
    return result

def trap(height):
    if not height:
        return 0
    n = len(height)
    if n < 3:
        return 0
    left_max = [0] * n
    left_max[0] = height[0]
    for i in range(1, n):
        left_max[i] = max(left_max[i-1], height[i])
    right_max = [0] * n
    right_max[-1] = height[-1]
    for i in range(n-2, -1, -1):
        right_max[i] = max(right_max[i+1], height[i])
    total = 0
    for i in range(n):
        total += min(left_max[i], right_max[i]) - height[i]
    return total
